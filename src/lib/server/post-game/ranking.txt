function rankingWinnerAccent(time){
 const t=Math.max(0,Math.min(1,(time-560)/420));
 const red=t*t*(3-2*t);
 const shake=t>0&&t<1?Math.sin(t*Math.PI*8)*5*Math.sin(t*Math.PI)*(1-t):0;
 return {red,shake};
}

 if(!players.length)return;
