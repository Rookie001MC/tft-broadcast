function motion(delay,dx=64,dy=0){
 const t=Math.max(0,Math.min(1,(animationTime-delay)/motionDuration));
 const eased=t<.5?4*t*t*t:1-Math.pow(-2*t+2,3)/2;
 ctx.save();ctx.globalAlpha*=eased;ctx.translate(dx*(1-eased),dy*(1-eased));
}
function stopAnimation(){cancelAnimationFrame(animationFrame);animationFrame=0}
function playAnimation(){
 stopAnimation();
 const started=performance.now();
 const units=Math.max(11,...data.players.map(p=>(p.boardPieces||[]).length));
 const duration=initial.ranking?1000+Math.max(0,data.players.length-2)*80+motionDuration:Math.max(0,data.players.length-1)*100+100+(units-1)*24+motionDuration;

 function tick(now){
  animationTime=(now-started)*1.2;draw();
  if(animationTime<duration){animationFrame=requestAnimationFrame(tick)}
  else{animationTime=Infinity;draw();animationFrame=0;exportButton.disabled=false;window.overlayState.animating=false}
 }
 tick(started);
}
const emptySlotUrl='https://raw.communitydragon.org/latest/plugins/rcp-fe-lol-tft-team-planner/global/default/images/team_codes_champion_empty.png';
window.overlayState={phase:'loading',loaded:0,total:0,missing:[]};
if(new URLSearchParams(location.search).has('obs')) document.body.classList.add('obs');
function text(s,x,y,size=18,color='#f3f0fa',weight=600,max=1000){ctx.fillStyle=color;ctx.font=`${weight} ${size}px "IBM Plex Sans",Arial,sans-serif`;s=String(s);while(ctx.measureText(s).width>max&&s.length>1)s=s.slice(0,-2)+'…';ctx.fillText(s,x,y)}
function box(x,y,w,h,color){ctx.fillStyle=color;ctx.fillRect(x,y,w,h)}
function image(path,x,y,w,h){const im=images.get(path);if(im)ctx.drawImage(im,x,y,w,h);return !!im}
function iconPaths(value){if(initial.ranking){value={players:[...value.players].sort((a,b)=>a.ffaStanding-b.ffaStanding).slice(0,1)}}const paths=new Set();if(initial.ranking&&winnerPortrait)paths.add(winnerPortrait);if(!initial.ranking&&value.players.some(p=>(p.boardPieces||[]).length<11))paths.add(emptySlotUrl);for(const p of value.players)for(const u of p.boardPieces||[]){paths.add(u.icon||'');mappedUrls.set(u.icon,initial.catalog?.champions?.[u.championId]||[]);for(const item of u.items||[]){paths.add(item.icon||'');mappedUrls.set(item.icon,initial.catalog?.items?.[item.nameId]||[])}}return paths}
function assetUrls(path){
 if(!path)return [];
 if(/^data:image\/(png|webp);base64,/.test(path))return [path];
 if(/^https:\/\/raw\.communitydragon\.org\//i.test(path))return [path];
 const prefix='/lol-game-data/assets/';if(!path.startsWith(prefix))return [];
 const relative=path.slice(prefix.length).toLowerCase();
 if(relative.split('/').includes('..'))return [];
 // LCU supplies the exact asset identity; do not match items by id=0 or translated names.
 return [...new Set([...(mappedUrls.get(path)||[]),...['latest','pbe'].flatMap(version=>[
  `https://raw.communitydragon.org/${version}/plugins/rcp-be-lol-game-data/global/default/${relative}`,
  `https://raw.communitydragon.org/${version}/game/${relative}`
 ])])];
}
async function loadImage(path){
 if(images.has(path))return true;
 for(const url of assetUrls(path)){
  const im=await new Promise(resolve=>{
   const img=new Image();img.crossOrigin='anonymous';
   const timer=setTimeout(()=>finish(null),12000);
   function finish(result){clearTimeout(timer);img.onload=null;img.onerror=null;resolve(result)}
   img.onload=async()=>{try{await img.decode();finish(img)}catch{finish(null)}};
   img.onerror=()=>finish(null);img.src=url;
  });
  if(im){images.set(path,im);return true}
 }
 return false;
}

 players.forEach((p,i)=>{
  const y=70+i*120, rank=p.ffaStanding??p.rank??i+1;

  const name=p.riotIdGameName||p.summonerName||'—';text(name,190,y+66,27,'#25212b',600,303);
  ctx.restore();
  const units=p.boardPieces||[],slots=Math.max(11,units.length),gap=8,size=Math.min(110,(1300-(slots-1)*gap)/slots);
  for(let j=0;j<slots;j++){
   const x=530+j*(size+gap),u=units[j];
   motion(i*100+100+j*24,48);
   const color=u?palette[Math.min(6,Math.max(0,Number(u.price)||0))]:'#343139';box(x,y,size,size,color);
   if(!u){image(emptySlotUrl,x+2,y+2,size-4,size-4);ctx.restore();continue}
   const portrait=images.get(u.icon);if(portrait){const side=Math.min(portrait.naturalWidth,portrait.naturalHeight);ctx.drawImage(portrait,(portrait.naturalWidth-side)/2,(portrait.naturalHeight-side)/2,side,side,x+2,y+2,size-4,size-4)}
   ctx.textAlign='center';ctx.strokeStyle='#181020';ctx.lineWidth=3;ctx.font='bold 18px "IBM Plex Sans"';const stars='★'.repeat(Math.max(0,Math.min(4,Number(u.level)||0)));ctx.strokeText(stars,x+size/2,y+13);text(stars,x+size/2,y+13,18,Number(u.level)>=3?'#ffdc6c':'#e7dff3',700);ctx.textAlign='left';
   const items=u.items||[],itemSize=Math.min(27,(size-8)/Math.max(3,items.length));items.forEach((item,k)=>{const ix=x+(size-items.length*itemSize)/2+k*itemSize,iy=y+size-itemSize;box(ix,iy,itemSize,itemSize,'#15101d');image(item.icon,ix+1,iy+1,itemSize-2,itemSize-2)});
   ctx.restore();
  }
 });
 ctx.restore();
 status.textContent=`1920 × 1080 · Nền xuất luôn trong suốt. ${players.length} người chơi${missing.size?` · ${missing.size} icon chưa có: đang hiển thị tên thay thế`:''}.`;
}
async function render(){
 stopAnimation();
 const token=++generation, paths=[...iconPaths(data)], failed=new Set();let completed=0;
 ready=false;exportButton.disabled=true;canvas.style.visibility='hidden';
 window.overlayState={phase:'loading',loaded:0,total:paths.length,missing:[]};

 let cursor=0;

}
